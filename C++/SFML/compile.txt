windres src/icon.rc -o src/icon.o

# CMake
mkdir&cd dist
cmake -S . -B dist -G Ninja -DCMAKE_BUILD_TYPE=Release
cmake --build dist

# MinGW
g++ -O2 -std=c++17 \
    src/*.cpp \
    src/icon.o \
    -o dist/LanguagePerformance.exe \
    -I/ucrt64/include \
    -L/ucrt64/lib \
    -lsfml-graphics -lsfml-window -lsfml-system

# Window

g++ -O2 -std=c++17 \
    src/*.cpp \
    src/icon.o \
    -o dist/LanguagePerformance.exe \
    -I/ucrt64/include \
    -L/ucrt64/lib \
    -mwindows \
    -lsfml-graphics -lsfml-window -lsfml-system

# Max Performance

g++ -O3 -ffast-math -mavx -mfma -fopenmp -flto -std=c++17 \
    src/*.cpp \
    src/icon.o \
    -o dist/LanguagePerformance.exe \
    -I/ucrt64/include \
    -L/ucrt64/lib \
    -mwindows \
    -lsfml-graphics -lsfml-window -lsfml-system

# Only your computer can launcher

g++ -O3 -march=native -ffast-math -mavx -mfma -fopenmp -flto -fprofile-generate -std=c++17 \
    src/*.cpp \
    src/icon.o \
    -o dist/LanguagePerformance.exe \
    -I/ucrt64/include \
    -L/ucrt64/lib \
    -mwindows \
    -lsfml-graphics -lsfml-window -lsfml-system

    -o dist/LanguagePerformance.exe \
    -I/ucrt64/include \
    -L/ucrt64/lib \
