windres src/icon.rc -O coff -o src/icon.res

#CMake
mkdir&cd dist
cmake -G Ninja -DCMAKE_BUILD_TYPE=Release ..
ninja

#MinGW
g++ -O2 -std=c++17 \
    src/*.cpp \
    src/icon.res \
    -o dist/LanguagePerformance.exe \
    -I/mingw64/include \
    -L/mingw64/lib \
    -lraylib -lopengl32 -lgdi32 -lwinmm

# Window

g++ -O2 -std=c++17 \
    src/*.cpp \
    src/icon.res \
    -o dist/LanguagePerformance.exe \
    -I/mingw64/include \
    -L/mingw64/lib \
    -mwindows \
    -lraylib -lopengl32 -lgdi32 -lwinmm

# Max Performance

g++ -O3 -ffast-math -mavx -mfma -fopenmp -flto -std=c++17 \
    src/*.cpp \
    src/icon.o \
    -o dist/LanguagePerformance.exe \
    -I/ucrt64/include \
    -L/ucrt64/lib \
    -mwindows \
    -lraylib -lopengl32 -lgdi32 -lwinmm

# Only your computer can launcher

g++ -O3 -march=native -ffast-math -mavx -mfma -fopenmp -flto -fprofile-generate -std=c++17 \
    src/*.cpp \
    src/icon.o \
    -o dist/LanguagePerformance.exe \
    -I/ucrt64/include \
    -L/ucrt64/lib \
    -mwindows \
    -lraylib -lopengl32 -lgdi32 -lwinmm

g++ -O3 -march=native -ffast-math -mavx -mfma -fopenmp -flto -fprofile-use -fprofile-correction -std=c++17 \
    src/*.cpp \
    src/icon.o \
    -o dist/LanguagePerformance.exe \
    -I/ucrt64/include \
    -L/ucrt64/lib \
    -mwindows \
    -lraylib -lopengl32 -lgdi32 -lwinmm
